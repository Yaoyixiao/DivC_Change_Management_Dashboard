<script>
  (function(){
    var card=document.getElementById('tx'), tbody=card.querySelector('.tx-tbody');
    var foot=card.querySelector('[data-foot]'), total=card.querySelector('[data-total]'), search=card.querySelector('.tx-search input');
    function rows(){ return [].slice.call(tbody.querySelectorAll('tr')); }
    function money(n){ return '$'+n.toLocaleString('en-US',{minimumFractionDigits:2,maximumFractionDigits:2}); }
    function refresh(){ var vis=rows().filter(function(r){return r.style.display!=='none';}); var sum=0; vis.forEach(function(r){ sum+=parseFloat(r.querySelector('.col-amt').dataset.val)||0; }); if(foot)foot.textContent=vis.length+(vis.length===1?' transaction':' transactions'); if(total)total.textContent=money(sum)+' total'; }
    card.querySelectorAll('th[data-sort]').forEach(function(th){ th.addEventListener('click', function(){ var key=th.dataset.sort, num=th.dataset.type==='num', dir=th.getAttribute('aria-sort')==='ascending'?'descending':'ascending'; card.querySelectorAll('th[data-sort]').forEach(function(o){ o.removeAttribute('aria-sort'); }); th.setAttribute('aria-sort',dir); rows().sort(function(a,b){ var av=a.querySelector('.col-'+key).dataset.val, bv=b.querySelector('.col-'+key).dataset.val; var c=num?(parseFloat(av)-parseFloat(bv)):(av.toLowerCase()<bv.toLowerCase()?-1:av.toLowerCase()>bv.toLowerCase()?1:0); return dir==='ascending'?c:-c; }).forEach(function(r){ tbody.appendChild(r); }); }); });
    search.addEventListener('input', function(){ var q=search.value.trim().toLowerCase(); rows().forEach(function(r){ r.style.display=r.textContent.toLowerCase().indexOf(q)>-1?'':'none'; }); refresh(); });
    refresh();
  })();
</script>
